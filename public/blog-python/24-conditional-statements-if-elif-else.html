# Python Conditional Statements
score = 85

if score >= 90:
    print("Grade: A+ 🌟")
elif score >= 80:
    print("Grade: A ✨")
elif score >= 70:
    print("Grade: B 👍")
else:
    print("Grade: Needs Improvement")
