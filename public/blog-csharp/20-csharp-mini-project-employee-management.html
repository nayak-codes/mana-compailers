అభినందనలు! మీరు C# బేసిక్స్, OOP సూత్రాలు, జెనెరిక్ కలెక్షన్స్, ఎక్సెప్షన్ హ్యాండ్లింగ్ మరియు LINQ వరకు C# ప్రయాణాన్ని విజయవంతంగా పూర్తి చేశారు. ఇప్పుడు మనం నేర్చుకున్న ప్రతి ఒక్క సాంకేతిక అంశాన్ని ఏకీకృతం చేస్తూ ఒక సంపూర్ణ ఎంటర్‌ప్రైజ్-గ్రేడ్ Employee Management System (క్యాప్‌స్టోన్ మినీ ప్రాజెక్ట్) ను నిర్మిద్దాం. ఇందులో డొమైన్ మోడలింగ్, CRUD ఆపరేషన్స్, LINQ అనలిటిక్స్ మరియు System.Text.Json ద్వారా డిస్క్‌పై డేటాను శాశ్వతంగా భద్రపరిచే (Persistence) రియల్-వరల్డ్ ఆర్కిటెక్చర్ ఉంటుంది!
1 ప్రాజెక్ట్ ఆర్కిటెక్చర్ & డొమైన్ మోడల్స్ డిజైన్

ఈ క్యాప్‌స్టోన్ ప్రాజెక్ట్‌లో మనం పాటించే ప్రొఫెషనల్ ఆర్కిటెక్చరల్ పద్ధతులు:

  • Department (Enum): ఐటీ, హెచ్ఆర్, ఫైనాన్స్, సేల్స్ వంటి స్థిరమైన విభాగాలు.
  • Employee (Class): ఐడీ, పేరు, డిపార్ట్‌మెంట్, మరియు జీతం లక్షణాలతో కూడిన ఎన్‌క్యాప్సులేటెడ్ మోడల్.
  • EmployeeSummary (Record): DTO (Data Transfer Object) మరియు రిపోర్టింగ్ కోసం ఇమ్మ్యూటబుల్ రికార్డ్.
  • IEmployeeRepository (Interface): డికప్లింగ్ మరియు డిపెండెన్సీ ఇంజెక్షన్ పునాది.
  • System.Text.Json: అప్లికేషన్ రీస్టార్ట్ అయినా డేటా పోకుండా employees.json ఫైల్‌లో భద్రపరచడం.
DomainModels.cs ⚡ Try in C# Compiler 
using System;

// Department Enum
public enum Department
{
    IT,
    HR,
    Finance,
    Sales
}

// Domain Model
public class Employee
{
    public int Id { get; init; }
    public string Name { get; set; } = string.Empty;
    public Department Dept { get; set; }
    public decimal Salary { get; set; }

    public Employee() { }

    public Employee(int id, string name, Department dept, decimal salary)
    {
        Id = id;
        Name = name;
        Dept = dept;
        Salary = salary;
    }

    public override string ToString() =>
        $"ID: {Id,4} | Name: {Name,-15} | Dept: {Dept,-8} | Salary: {Salary:C}";
}

// Immutable Summary Record for Analytics
public record DepartmentSummary(string Department, int HeadCount, decimal AvgSalary, decimal MaxSalary);
2 Repository Interface & Service Implementation

డేటాను ప్రాసెస్ చేసే బిజినెస్ లాజిక్‌ను కన్సోల్ యూజర్ ఇంటర్‌ఫేస్ నుండి వేరు చేయడం క్లీన్ కోడ్ సూత్రం. దీనికోసం IEmployeeService ఇంటర్‌ఫేస్ మరియు దాని ఇంప్లిమెంటేషన్‌ను పరిశీలిద్దాం:

EmployeeService.cs ⚡ Try in C# Compiler 
using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Text.Json;

public class EmployeeService
{
    private List _employees = new List();
    private readonly string _filePath;

    public EmployeeService(string filePath = "employees.json")
    {
        _filePath = filePath;
        LoadFromFile();
    }

    public bool AddEmployee(Employee emp)
    {
        if (_employees.Any(e => e.Id == emp.Id))
            return false; // Duplicate ID

        _employees.Add(emp);
        SaveToFile();
        return true;
    }

    public List GetAll() => _employees.OrderBy(e => e.Id).ToList();

    public Employee? FindById(int id) => _employees.FirstOrDefault(e => e.Id == id);

    public bool DeleteEmployee(int id)
    {
        var emp = FindById(id);
        if (emp == null) return false;

        _employees.Remove(emp);
        SaveToFile();
        return true;
    }

    // LINQ Analytics: Group by Department
    public List GetDepartmentSummaries()
    {
        return _employees
            .GroupBy(e => e.Dept)
            .Select(g => new DepartmentSummary(
                g.Key.ToString(),
                g.Count(),
                g.Average(e => e.Salary),
                g.Max(e => e.Salary)
            ))
            .ToList();
    }

    private void SaveToFile()
    {
        try
        {
            var options = new JsonSerializerOptions { WriteIndented = true };
            string json = JsonSerializer.Serialize(_employees, options);
            File.WriteAllText(_filePath, json);
        }
        catch (Exception ex)
        {
            Console.WriteLine($"[Storage Error]: {ex.Message}");
        }
    }

    private void LoadFromFile()
    {
        if (!File.Exists(_filePath)) return;
        try
        {
            string json = File.ReadAllText(_filePath);
            _employees = JsonSerializer.Deserialize>(json) ?? new List();
        }
        catch
        {
            _employees = new List();
        }
    }
}
3 ఇంటరాక్టివ్ కన్సోల్ UI మెనూ & ఎగ్జిక్యూషన్ లాజిక్

యూజర్ ఫ్రెండ్లీ కమాండ్-లైన్ ఇంటర్‌ఫేస్ (CLI) ద్వారా ఉద్యోగులను జోడించడం, జాబితాను చూడటం, డిపార్ట్‌మెంట్ వారీగా LINQ రిపోర్టులను వీక్షించడం ఎలాగో ఈ క్రింది పూర్తి కోడ్‌లో చూడవచ్చు:

Program.cs (Complete Application) ⚡ Try in C# Compiler 
using System;
using System.Collections.Generic;

class Program
{
    static void Main()
    {
        EmployeeService service = new EmployeeService("company_data.json");

        // Seed initial data if empty
        if (service.GetAll().Count == 0)
        {
            service.AddEmployee(new Employee(101, "Ravi Teja", Department.IT, 75000m));
            service.AddEmployee(new Employee(102, "Sneha Latha", Department.HR, 52000m));
            service.AddEmployee(new Employee(103, "Manoj Kumar", Department.IT, 82000m));
            service.AddEmployee(new Employee(104, "Priya Sharma", Department.Finance, 68000m));
        }

        bool running = true;
        while (running)
        {
            Console.WriteLine("
========================================");
            Console.WriteLine("🏢 EMPLOYEE MANAGEMENT SYSTEM (.NET 8)");
            Console.WriteLine("========================================");
            Console.WriteLine("1. View All Employees");
            Console.WriteLine("2. Add New Employee");
            Console.WriteLine("3. Search Employee by ID");
            Console.WriteLine("4. View Department Analytics (LINQ)");
            Console.WriteLine("5. Delete Employee");
            Console.WriteLine("6. Exit");
            Console.Write("Enter your choice (1-6): ");

            string? choice = Console.ReadLine();
            Console.WriteLine();

            switch (choice)
            {
                case "1":
                    var list = service.GetAll();
                    Console.WriteLine($"Total Employees: {list.Count}");
                    list.ForEach(e => Console.WriteLine(e));
                    break;

                case "2":
                    AddNewEmployeeUI(service);
                    break;

                case "3":
                    Console.Write("Enter Employee ID: ");
                    if (int.TryParse(Console.ReadLine(), out int searchId))
                    {
                        var found = service.FindById(searchId);
                        Console.WriteLine(found != null ? found.ToString() : "Employee not found!");
                    }
                    break;

                case "4":
                    var summaries = service.GetDepartmentSummaries();
                    Console.WriteLine("--- Department Analytics (LINQ GroupBy) ---");
                    foreach (var s in summaries)
                    {
                        Console.WriteLine($"Dept: {s.Department,-8} | Count: {s.HeadCount} | Avg: {s.AvgSalary:C} | Max: {s.MaxSalary:C}");
                    }
                    break;

                case "5":
                    Console.Write("Enter ID to delete: ");
                    if (int.TryParse(Console.ReadLine(), out int delId))
                    {
                        bool deleted = service.DeleteEmployee(delId);
                        Console.WriteLine(deleted ? "Deleted successfully!" : "ID not found!");
                    }
                    break;

                case "6":
                    Console.WriteLine("Thank you for using the application! Goodbye.");
                    running = false;
                    break;

                default:
                    Console.WriteLine("Invalid choice! Please select 1-6.");
                    break;
            }
        }
    }

    static void AddNewEmployeeUI(EmployeeService service)
    {
        Console.Write("Enter Employee ID: ");
        if (!int.TryParse(Console.ReadLine(), out int id)) { Console.WriteLine("Invalid ID!"); return; }

        Console.Write("Enter Name: ");
        string name = Console.ReadLine() ?? "Unknown";

        Console.Write("Select Dept (0=IT, 1=HR, 2=Finance, 3=Sales): ");
        if (!Enum.TryParse(Console.ReadLine(), out Department dept)) dept = Department.IT;

        Console.Write("Enter Monthly Salary: ");
        if (!decimal.TryParse(Console.ReadLine(), out decimal salary)) salary = 30000m;

        bool added = service.AddEmployee(new Employee(id, name, dept, salary));
        Console.WriteLine(added ? "Employee added successfully and saved to JSON!" : "Failed: ID already exists!");
    }
}
