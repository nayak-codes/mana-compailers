⚡ Modern C (C99 / C11) 🟢 Chapter 20 of 20 📂 Phase 05: స్ట్రక్చర్స్, డైనమిక్ మెమరీ & ప్రాజెక్ట్ 📅 2026 Edition
📌 Covered in this chapter: Structs · Pointers · Dynamic Allocation · File Persistence తో పూర్తి స్థాయి రియల్-వరల్డ్ C ప్రాజెక్ట్!
అభినందనలు! మనం C ప్రోగ్రామింగ్ ప్రాథమిక వేరియబుల్స్ నుండి మొదలుకొని కంట్రోల్ ఫ్లో, లూప్స్, ఫంక్షన్స్, పాయింటర్లు, డైనమిక్ మెమరీ, మరియు ఫైల్ హ్యాండ్లింగ్ వరకు అన్ని కాన్సెప్ట్‌లను పూర్తి చేశాం. ఈ ముగింపు అధ్యాయంలో వాటన్నింటినీ ఒకచోట చేర్చి ఒక పూర్తి స్థాయి ప్రొడక్షన్-గ్రేడ్ స్టూడెంట్ రికార్డ్ మేనేజ్‌మెంట్ సిస్టమ్ (Student Record Management System) ను నిర్మిద్దాం.
1ప్రాజెక్ట్ ఆర్కిటెక్చర్ & కోర్ ఫీచర్లు

ఈ ప్రాజెక్ట్ ఒక పాఠశాల లేదా కళాశాల అడ్మినిస్ట్రేషన్ వ్యవస్థ లాగా పనిచేస్తుంది. దీనిలో ఉపయోగించిన ఆధునిక ఇంజనీరింగ్ పద్ధతులు:

1. Struct & Typedef రోల్ నంబర్, పేరు, మార్కులు కలిగిన శుభ్రమైన స్టూడెంట్ రికార్డ్ మోడల్.
2. Dynamic Realloc విద్యార్థులు పెరిగే కొద్దీ హీప్ మెమరీని ఆటోమేటిక్‌గా విస్తరించే డైనమిక్ అరే.
3. Binary File Persistence డేటాను students.dat ఫైల్‌లో భద్రపరచడం & రీలోడ్ చేయడం.
4. Interactive CLI Menu యూజర్ ఫ్రెండ్లీ కన్సోల్ మెనూ మరియు ఇన్‌పుట్ వాలిడేషన్.
2పూర్తి C సోర్స్ కోడ్ (Complete Production Code)
C — Student Record Management System ⚡ Try in C Compiler 
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define DATA_FILE "students.dat"

typedef struct {
    int rollNo;
    char name[40];
    float marks;
} Student;

// గ్లోబల్ లేదా మెయిన్ మేనేజ్డ్ పాయింటర్స్
Student *students = NULL;
int studentCount = 0;

void addStudent(void) {
    // మెమరీని 1 ఎలిమెంట్ పరిమాణం పెంచడం
    Student *temp = (Student*) realloc(students, (studentCount + 1) * sizeof(Student));
    if (temp == NULL) {
        printf("ఎర్రర్: మెమరీ సరిపోలేదు!\n");
        return;
    }
    students = temp;

    printf("\nవిద్యార్థి రోల్ నంబర్: ");
    scanf("%d", &students[studentCount].rollNo);
    printf("విద్యార్థి పేరు: ");
    scanf(" %39s", students[studentCount].name);
    printf("మార్కులు: ");
    scanf("%f", &students[studentCount].marks);

    studentCount++;
    printf("✅ విద్యార్థి రికార్డు విజయవంతంగా జోడించబడింది!\n");
}

void displayStudents(void) {
    if (studentCount == 0) {
        printf("\nరికార్డులు ఏవీ లేవు!\n");
        return;
    }
    printf("\n-------------------------------------------------\n");
    printf("%-10s %-25s %-10s\n", "రోల్ నెం", "పేరు", "మార్కులు");
    printf("-------------------------------------------------\n");
    for (int i = 0; i < studentCount; i++) {
        printf("%-10d %-25s %-10.2f\n", students[i].rollNo, students[i].name, students[i].marks);
    }
    printf("-------------------------------------------------\n");
}

void saveToFile(void) {
    FILE *fp = fopen(DATA_FILE, "wb");
    if (fp == NULL) {
        printf("ఫైల్ సేవ్ చేయడం సాధ్యం కాలేదు!\n");
        return;
    }
    fwrite(&studentCount, sizeof(int), 1, fp);
    fwrite(students, sizeof(Student), studentCount, fp);
    fclose(fp);
    printf("💾 సమాచారం డిస్క్‌పై సేవ్ చేయబడింది.\n");
}

void loadFromFile(void) {
    FILE *fp = fopen(DATA_FILE, "rb");
    if (fp == NULL) return; // ఫైల్ ఇంకా లేకపోతే ఏమీ చేయనవసరం లేదు

    fread(&studentCount, sizeof(int), 1, fp);
    students = (Student*) malloc(studentCount * sizeof(Student));
    if (students != NULL) {
        fread(students, sizeof(Student), studentCount, fp);
    }
    fclose(fp);
}

int main(void) {
    loadFromFile(); // గతంలో సేవ్ చేసిన రికార్డులను లోడ్ చేయడం
    int choice;

    do {
        printf("\n====== విద్యార్థి రికార్డ్ మేనేజ్‌మెంట్ ======\n");
        printf("1. కొత్త విద్యార్థిని జోడించు (Add)\n");
        printf("2. అన్ని రికార్డులను ప్రదర్శించు (Display All)\n");
        printf("3. సేవ్ చేసి నిష్క్రమించు (Save & Exit)\n");
        printf("మీ ఎంపిక (1-3): ");
        if (scanf("%d", &choice) != 1) break;

        switch (choice) {
            case 1: addStudent(); break;
            case 2: displayStudents(); break;
            case 3: saveToFile(); break;
            default: printf("చెల్లని ఎంపిక! దయచేసి మళ్లీ ప్రయత్నించండి.\n");
        }
    } while (choice != 3);

    // మెమరీ క్లీనప్
    if (students != NULL) {
        free(students);
        students = NULL;
    }
    printf("ధన్యవాదాలు! ప్రాజెక్ట్ విజయవంతంగా ముగిసింది.\n");
    return 0;
}
3లైన్-బై-లైన్ ఆర్కిటెక్చర్ విశ్లేషణ
  • realloc() డైనమిక్ గ్రోత్: యూజర్ కొత్త విద్యార్థిని యాడ్ చేసిన ప్రతిసారీ studentCount + 1 తో హీప్ మెమరీని సరిగ్గా అవసరమైనంత మాత్రమే విస్తరిస్తుంది.
  • బైనరీ పర్సిస్టెన్స్: fwrite మరియు fread తో మొత్తం స్ట్రక్చర్ అరేను క్షణాల్లో ఫైల్‌లోకి రాయడం/చదవడం జరుగుతుంది.
  • జీరో మెమరీ లీక్స్: ప్రోగ్రామ్ ముగిసే ముందు free(students) తో కేటాయించిన హీప్ మెమరీని పూర్తిగా ఆపరేటింగ్ సిస్టమ్‌కు విడుదల చేశాం.
