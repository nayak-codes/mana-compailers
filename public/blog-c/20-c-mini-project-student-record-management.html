C — Student Records Project ▶ Run in Compiler 
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int roll;
    char name[40];
    float marks;
} Student;

void displayStudent(const Student* s) {
    printf("Roll: %-4d | Name: %-15s | Marks: %.2f\n", s->roll, s->name, s->marks);
}

int main() {
    int count = 2;
    Student* list = (Student*)malloc(count * sizeof(Student));
    if (!list) return 1;

    // Student 1
    list[0].roll = 101;
    strcpy(list[0].name, "Rajesh Kumar");
    list[0].marks = 92.5f;

    // Student 2
    list[1].roll = 102;
    strcpy(list[1].name, "Priya Sharma");
    list[1].marks = 95.0f;

    printf("--- STUDENT DIRECTORY (C CAPSTONE PROJECT) ---\n");
    for (int i = 0; i < count; ++i) {
        displayStudent(&list[i]);
    }

    free(list);
    list = NULL;
    return 0;
}
